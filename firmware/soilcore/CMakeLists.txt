cmake_minimum_required(VERSION 3.16)
project(soilcore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(soilcore core/soilcore.cpp)
target_include_directories(soilcore PUBLIC core)
add_executable(soilcore_cli platform/native/main.cpp)
target_link_libraries(soilcore_cli PRIVATE soilcore)
enable_testing()
add_executable(soilcore_tests tests/tests.cpp)
target_link_libraries(soilcore_tests PRIVATE soilcore)
# Keep assertions active even when the firmware is built in Release mode.
if(MSVC)
  target_compile_options(soilcore_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(soilcore_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME soilcore_tests COMMAND soilcore_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME soilcore_cli_recovery COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_recovery.py" "$<TARGET_FILE:soilcore_cli>")
endif()
